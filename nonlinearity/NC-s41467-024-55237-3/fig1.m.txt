%% Floquet topological soliton reproduction
% Based on the model and self-consistency procedure in the paper.
clear; close all; clc

Nx=14; Ny=14; N=Nx*Ny;
z0=1; nstep=4; dz=z0/nstep;
Lambda=1.85; J=Lambda/dz;                 % integral J dz = Lambda
Pscan=linspace(0.02,0.20,40);              % renormalized power

% Four bond matchings used during the four driving quarters
Hcell=cell(1,4);
for m=1:4
    Hcell{m}=bond_hamiltonian(Nx,Ny,m,J);
end

%% Figure 2b-e and 2f: one converged soliton and micromotion
P0=0.088;
[phi,eps,err]=self_consistent(P0,Hcell,Nx,Ny);
profiles=zeros(N,4); psi=phi;
for q=1:4
    profiles(:,q)=abs(psi).^2/P0;
    psi=step_nonlinear(psi,Hcell{q},dz);
end
figure('Color','w','Name','Fig2b-e');
for q=1:4
    subplot(1,4,q)
    imagesc(reshape(profiles(:,q),Ny,Nx)); axis image off
    set(gca,'ColorScale','log'); caxis([1e-6 1]); colormap(jet)
    title(char('B'+q-1));
end

% Four sites around the launch site, used for Fig. 2f
c=sub2ind([Ny,Nx],round(Ny/2),round(Nx/2));
site=[c,c+1,c+Ny,c-1];
zplot=linspace(0,1,401); I4=zeros(4,numel(zplot));
psi=phi; I4(:,1)=abs(psi(site)).^2/P0;
for k=1:400
    q=mod(k-1,4)+1;
    psi=step_nonlinear(psi,Hcell{q},dz/100);
    if mod(k,100)==0
        I4(:,k+1)=abs(psi(site)).^2/P0;
    else
        I4(:,k+1)=I4(:,k);
    end
end
figure('Color','w','Name','Fig2f');
plot(zplot,I4,'LineWidth',1.6); xlabel('z/z_0'); ylabel('Normalized intensity');
legend('1','2','3','4'); box on

%% Figure 2g-h and Figure 9a: power scan
IPR=zeros(size(Pscan)); E=zeros(size(Pscan)); sol=phi;
for k=1:numel(Pscan)
    [sol,E(k)]=self_consistent(Pscan(k),Hcell,Nx,Ny,sol);
    I=abs(sol).^2; IPR(k)=sum(I.^2)/(sum(I)^2);
end
figure('Color','w','Name','Fig2h-Fig9a');
plot(Pscan,IPR,'o-','Color',[0.90 0.45 0.45]);
xlabel('P (mm^{-1})'); ylabel('Localization, IPR'); box on

%% Figure 9b-d: normalized intensity at three powers
P9=[0.04 0.12 0.18];
figure('Color','w','Name','Fig9b-d');
for k=1:3
    [s,~]=self_consistent(P9(k),Hcell,Nx,Ny);
    subplot(1,3,k); imagesc(reshape(abs(s).^2/P9(k),Ny,Nx));
    axis image off; set(gca,'ColorScale','log'); caxis([1e-6 1]);
    title(sprintf('P = %.3f',P9(k)));
end

%% Figure 8: self-consistency convergence
[P8,eps8,err8]=self_consistent(0.088,Hcell,Nx,Ny); %#ok<ASGLU>
% To display convergence, rerun while recording quasienergy and error
[~,qhist,ehist]=self_consistent_history(0.088,Hcell,Nx,Ny);
figure('Color','w','Name','Fig8');
subplot(1,2,1); plot(1:numel(qhist),qhist,'bo-'); xlabel('Iteration'); ylabel('Quasienergy'); box on
subplot(1,2,2); semilogy(1:numel(ehist),ehist,'ro-'); xlabel('Iteration'); ylabel('log_{10}(Error)'); box on

%% ---------- local functions ----------
function H=bond_hamiltonian(Nx,Ny,m,J)
N=Nx*Ny; H=sparse(N,N);
for x=1:Nx
 for y=1:Ny
  % cyclic nearest-neighbour matching; each quarter activates one bond orientation
  if m==1 && mod(x+y,2)==0, addbond(x,y,x+1,y); end
  if m==2 && mod(x+y,2)==0, addbond(x,y,x,y+1); end
  if m==3 && mod(x+y,2)==1, addbond(x,y,x-1,y); end
  if m==4 && mod(x+y,2)==1, addbond(x,y,x,y-1); end
 end
end
 function addbond(x1,y1,x2,y2)
  x2=mod(x2-1,Nx)+1; y2=mod(y2-1,Ny)+1;
  a=sub2ind([Ny,Nx],y1,x1); b=sub2ind([Ny,Nx],y2,x2);
  H(a,b)=J; H(b,a)=J;
 end
end

function out=step_nonlinear(psi,H,dz)
% Symmetric split-step approximation to i dpsi/dz=(H-|psi|^2)psi
D=spdiags(-abs(psi).^2,0,length(psi),length(psi));
out=expmv_k((H+D),psi,dz);
end

function y=expmv_k(A,x,t)
% Krylov-free small-step approximation; repeated Euler midpoint steps
nsub=8; h=t/nsub; y=x;
for k=1:nsub
    k1=-1i*A*y; ym=y+0.5*h*k1;
    k2=-1i*A*ym; y=y+h*k2;
end
end

function [phi,eps,err]=self_consistent(P,Hcell,Nx,Ny,varargin)
N=Nx*Ny;
if nargin>=5, phi=varargin{1}; else, phi=zeros(N,1); phi(sub2ind([Ny,Nx],round(Ny/2),round(Nx/2)))=sqrt(P); end
phi=phi*sqrt(P/sum(abs(phi).^2)); eps=0; err=1;
for it=1:120
 old=phi; psi=phi;
 for q=1:4, psi=step_nonlinear(psi,Hcell{q},0.25); end
 ov=(old'*psi)/(sqrt(sum(abs(old).^2))*sqrt(sum(abs(psi).^2)));
 phi=psi*sqrt(P/sum(abs(psi).^2));
 eps=-angle(ov); err=sum(abs(phi-old).^2)/P;
 if err<1e-10, break; end
end
end

function [phi,qhist,ehist]=self_consistent_history(P,Hcell,Nx,Ny)
N=Nx*Ny; phi=zeros(N,1); phi(sub2ind([Ny,Nx],round(Ny/2),round(Nx/2)))=sqrt(P); qhist=[]; ehist=[];
for it=1:120
 old=phi; psi=phi; for q=1:4, psi=step_nonlinear(psi,Hcell{q},0.25); end
 ov=(old'*psi)/(norm(old)*norm(psi)); phi=psi*sqrt(P/sum(abs(psi).^2));
 qhist(end+1)=-angle(ov); ehist(end+1)=sum(abs(phi-old).^2)/P;
 if ehist(end)<1e-10, break; end
end
end
